cmake_minimum_required(VERSION 3.16)
project(FM_Tracker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nuked_opl3 STATIC vendor/opl3.c)
if(WIN32)
    add_executable(FM_Tracker WIN32 FM_Win32.cpp)
    target_compile_definitions(FM_Tracker PRIVATE UNICODE _UNICODE)
    target_link_libraries(FM_Tracker PRIVATE nuked_opl3 user32 gdi32 winmm comdlg32 shell32 avrt)
    if(MSVC)
        target_compile_options(FM_Tracker PRIVATE /utf-8 /W4)
        target_link_options(FM_Tracker PRIVATE /STACK:8388608)
    elseif(MINGW)
        target_link_options(FM_Tracker PRIVATE -municode -Wl,--stack,8388608)
    endif()
endif()
add_executable(FM_tests tests.cpp)
target_link_libraries(FM_tests PRIVATE nuked_opl3)
if(MSVC)
    target_compile_options(FM_tests PRIVATE /UNDEBUG)
    target_link_options(FM_tests PRIVATE /STACK:8388608)
else()
    target_compile_options(FM_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME FM_core COMMAND FM_tests)
